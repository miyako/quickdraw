cmake_minimum_required(VERSION 3.16)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum supported macOS")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(component-cli LANGUAGES CXX)

add_executable(component-cli main.cpp)
target_compile_features(component-cli PRIVATE cxx_std_17)
set_target_properties(component-cli PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(component-cli PRIVATE /utf-8)
elseif(MINGW)
  target_link_options(component-cli PRIVATE -municode)
endif()
